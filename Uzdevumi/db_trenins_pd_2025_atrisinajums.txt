-- 1. Izveidojam Kategorijas
CREATE TABLE Kategorijas (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nosaukums TEXT NOT NULL
);

-- 2. Izveidojam Preces (Saite uz Kategoriju)
CREATE TABLE Preces (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nosaukums TEXT NOT NULL,
    cena REAL NOT NULL,
    kategorijas_id INTEGER,
    FOREIGN KEY (kategorijas_id) REFERENCES Kategorijas(id)
);

-- 3. Izveidojam Klientus
CREATE TABLE Klienti (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    vards TEXT NOT NULL,
    uzvards TEXT NOT NULL,
    epasts TEXT UNIQUE
);

-- 4. Izveidojam Piegādātājus
CREATE TABLE Piegadataji (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    uznemums TEXT NOT NULL,
    telefons TEXT
);

-- 5. Izveidojam Pasūtījumus (Saites uz Klientu un Piegādātāju)
CREATE TABLE Pasutijumi (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    datums TEXT,
    klienta_id INTEGER,
    piegadataja_id INTEGER,
    FOREIGN KEY (klienta_id) REFERENCES Klienti(id),
    FOREIGN KEY (piegadataja_id) REFERENCES Piegadataji(id)
);

-- 6. STARPTABULA (Many-to-Many starp Pasūtījumiem un Precēm)
CREATE TABLE PasutijumaPreces (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    pasutijuma_id INTEGER,
    preces_id INTEGER,
    daudzums INTEGER DEFAULT 1,
    FOREIGN KEY (pasutijuma_id) REFERENCES Pasutijumi(id),
    FOREIGN KEY (preces_id) REFERENCES Preces(id)
);
